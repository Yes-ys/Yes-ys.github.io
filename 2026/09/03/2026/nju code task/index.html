class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
       int dp0 = 0,dp1 = 0,dpi = 0;
       for(int i = 2;i <= cost.size();i++){
            dpi = min(dp0 + cost[i-2],dp1 + cost[i-1]);
            dp0 = dp1;
            dp1 = dpi;
       }
        return dpi;
    }
};

动态规划转移方程:dp[i] = min(dp[i-1]+cost[i-1],dp[i-2]+cost[i-2])

class Solution {
public:
    int dp[100][100];
    int uniquePaths(int m, int n) {
        for(int i = 0;i < m;i++)dp[i][0] = 1;
        for(int j = 0;j < n;j++)dp[0][j] = 1;
        for(int i = 1;i < m;i++){
            for(int j = 1;j < n;j++){
                dp[i][j] = dp[i-1][j]+dp[i][j-1];
            }
        }
        return dp[m-1][n-1];
    }
};

dp[i][j] = dp[i-1][j]+dp[i][j-1];

到达(i,j)的路径数量等于到达(i-1,j)的路径数量+到达(i,j-1)的路径数量

class Solution {
public:
int dp[59];
int integerBreak(int n) {
dp[2] = 1;
for(int i = 3;i <= n;i++){
for(int j = 1;j <= i-2;j++){
dp[i] = max(dp[i],max((i-j)*j,dp[i-j]*j));
}
}
return dp[n];
}
};

  1. dp[i]代表将i拆分后的最大乘积；
  2. (i-j)*j对应将i拆分成两个数，dp[i-j] * j对应将i拆分成两个以上的数

找到所有二叉搜索树

二叉搜索树：

  1. 若它的左子树不空，则左子树上所有结点的值均小于它的根结点的值；
  2. 若它的右子树不空，则右子树上所有结点的值均大于它的根结点的值；
  3. 它的左、右子树也分别为二叉搜索树
class Solution {
public:
    int dp[20];
    int numTrees(int n) {
        dp[0] = 1;//0个节点，1个二叉搜索树
        for(int i = 1;i <= n;i++){
            for(int j = 0;j < i;j++){
                dp[i] += dp[j]*dp[i-1-j];
            }
        }
        return dp[n];
    }
};
  1. dp[i]，i个节点时能构成的二叉搜索树的数量
  2. 转移方程：dp[i] = sum dp[j]*dp[i-1-j], j from 0 to i-1

仅考虑各种数量节点下，二叉搜索树的形态，很容易发现该规律

class Solution {
public:
    int change(int amount, vector<int>& coins) {
        int bagSize = amount;

        vector<vector<uint64_t>> dp(coins.size(), vector<uint64_t>(bagSize + 1, 0));

        // 初始化最上行，整除的时候有一种方案
        for (int j = 0; j <= bagSize; j++) {
            if (j % coins[0] == 0) dp[0][j] = 1;//dp[0][0]是不用考虑的，没有意义
        }
        // 初始化最左列，注意都有一种方法，那就是不选任何硬币，对应和为0
        for (int i = 0; i < coins.size(); i++) {
            dp[i][0] = 1; 
        }
        // 以下遍历顺序行列可以颠倒
        for (int i = 1; i < coins.size(); i++) { // 行，遍历物品
            for (int j = 0; j <= bagSize; j++) { // 列，遍历背包
                if (coins[i] > j) dp[i][j] = dp[i - 1][j]; 
                else dp[i][j] = dp[i - 1][j] +  dp[i][j - coins[i]];//0-1背包的目标和变种（加法原理） + 完全背包
            }
        }
        return dp[coins.size() - 1][bagSize];
    }
};

we set two points in the left-most and right-most

we move the right-most point along the left. If the point’s element is less than the pivot we swap the elemnt of right-most point and left-most point

In details, we will contain a dp arrary to present the shortest distance between any two points on the graph.

The array has two dimension. The first dimension is the start point’s serial number and the second dimension is the end point’s serial number.

We will intialize the array. If two points is directly conected, we can set their distance in the array. And for the same point, the distance in the arrary is 0.

We use a tripe loop to solve this dynamic programming. In the first loop, we enumerate the intermediate point between the start point and tne end point. In the second loop we eunmerate the start point and the end point in the third loop.
